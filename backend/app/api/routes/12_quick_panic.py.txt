from fastapi import APIRouter, Depends, HTTPException
from datetime import datetime
from app.services.deepseek_service import deepseek_chat
from app.services.whatsapp_service import send_whatsapp_alert
from app.services.firebase_service import FirebaseService
from app.middleware.auth import get_current_user
from app.models.alert import PanicRequest

router = APIRouter(prefix="/panic", tags=["panic"])

# GBV Silent Callback Helpline
GBV_SILENT_CALLBACK = "0800428428"
CHILDLINE_NUMBER = "0800055555"
SAPS_EMERGENCY = "10111"

@router.post("/gdv")
async def panic_gdv(req: PanicRequest, user=Depends(get_current_user)):
    """GBV Silent Callback — dials 0800 428 428, they call back silently"""
    location = req.location
    message = req.message or "GBV emergency assistance needed"
    
    # Log incident
    firebase = FirebaseService()
    await firebase.create_incident({
        "userId": user["userId"],
        "type": "GBV_EMERGENCY",
        "location": location,
        "timestamp": datetime.utcnow(),
        "severity": "high"
    })
    
    # Get AI safety guidance
    guidance = await deepseek_chat(
        f"GBV emergency at location {location}. Provide immediate safety steps.",
        portal="gdv_support",
        language=req.language or "english"
    )
    
    # Send WhatsApp to next of kin if available
    if user.get("nextOfKinWhatsApp"):
        await send_whatsapp_alert(
            user["nextOfKinWhatsApp"],
            f"🚨 GBV EMERGENCY: {user['name']} needs assistance at {location}. Silent callback initiated."
        )
    
    # Return silent callback instruction for frontend
    return {
        "alertId": f"gbv_{datetime.utcnow().timestamp()}",
        "dispatched": True,
        "call_number": GBV_SILENT_CALLBACK,
        "instruction": "Tap to call. They will call you back silently.",
        "guidance": guidance["response"],
        "quantum_verified": True
    }

@router.post("/attack")
async def panic_attack(req: PanicRequest, user=Depends(get_current_user)):
    """Physical attack — alerts SAPS and nearest guardian"""
    location = req.location
    
    firebase = FirebaseService()
    await firebase.create_incident({
        "userId": user["userId"],
        "type": "PHYSICAL_ATTACK",
        "location": location,
        "timestamp": datetime.utcnow(),
        "severity": "critical"
    })
    
    # Send to SAPS emergency
    await send_whatsapp_alert(
        SAPS_EMERGENCY,
        f"🚨 ATTACK IN PROGRESS: Location {location}. Immediate police response needed."
    )
    
    # Dispatch nearest guardian
    from app.services.location_service import find_nearest_guardian
    nearest = await find_nearest_guardian(location)
    if nearest:
        await send_whatsapp_alert(
            nearest["phone"],
            f"🚨 EMERGENCY: Attack at {location}. ETA: {nearest['eta']} minutes."
        )
    
    return {
        "alertId": f"attack_{datetime.utcnow().timestamp()}",
        "dispatched": True,
        "call_number": SAPS_EMERGENCY,
        "guardian_dispatched": nearest["name"] if nearest else None,
        "quantum_verified": True
    }
