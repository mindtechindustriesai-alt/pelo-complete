def get_current_user(credentials: HTTPAuthorizationCredentials = Depends(security)):
    token = credentials.credentials
    try:
        payload = jwt.decode(token, JWT_SECRET, algorithms=["HS256"])
        return {"userId": payload.get("sub"), "name": payload.get("name"), "email": payload.get("email")}
    except:
        raise HTTPException(status_code=401, detail="Invalid token")

# DEEPSEEK SERVICE (Offline Fallback)
# ============================================================
DEEPSEEK_URL = "https://api.deepseek.com/v1/chat/completions"

SYSTEM_PROMPTS = {
    "pelo_safety": """You are PELO Guardian AI, a community safety assistant for South Africa. Help with: incident reporting, emergency contacts, crime prevention. Emergency: SAPS 10111 | Ambulance 10177 | GBV 0800150150 | Silent Callback 0800428428 | Childline 0800055555. Quantum verified: CHSH S=2.76.""",
    "incident_classifier": """Respond ONLY with JSON: {"severity": "high|medium|low", "category": "theft|violence|suspicious|gdv|fire|other", "dispatch_guardian": true|false, "summary": "one sentence"}""",
    "gdv_support": """You are a compassionate GBV assistant. Provide immediate safety steps. Helplines: 0800 428 428 (silent callback), 0800 150 150.""",
    "recipe_generator": """Suggest budget recipes for South Africa under R50 using Boxer/Shoprite ingredients. Include cost. Respond in user's language.""",
}

async def deepseek_chat(message: str, portal: str = "pelo_safety", language: str = "english") -> Dict:
    if not DEEPSEEK_API_KEY:
        return {"response": offline_fallback(message), "source": "offline"}
    
    system = SYSTEM_PROMPTS.get(portal, SYSTEM_PROMPTS["pelo_safety"])
    if language != "english":
        system += f"\n\nRespond in {language}."
    
    try:
        async with httpx.AsyncClient(timeout=60.0) as client:
            r = await client.post(
                DEEPSEEK_URL,
                headers={"Authorization": f"Bearer {DEEPSEEK_API_KEY}", "Content-Type": "application/json"},
                json={"model": "deepseek-chat", "messages": [{"role": "system", "content": system}, {"role": "user", "content": message}], "temperature": 0.7, "max_tokens": 1024}
            )
            if r.status_code == 200:
                return {"response": r.json()["choices"][0]["message"]["content"], "source": "deepseek"}
            return {"response": offline_fallback(message), "source": "api_error"}
    except:
        return {"response": offline_fallback(message), "source": "exception"}

async def classify_incident(description: str) -> dict:
    result = await deepseek_chat(description, portal="incident_classifier")
    try:
        return json.loads(result["response"])
    except:
        return {"severity": "medium", "category": "other", "dispatch_guardian": True, "summary": description[:100]}

def offline_fallback(message: str) -> str:
    m = message.lower()
    if "gdv" in m or "gbv" in m:
        return "GBV Helpline: 0800 150 150 | Silent Callback: 0800 428 428. You are not alone."
    if "child" in m:
        return "Childline SA: 0800 055 555. Report immediately."
    if "robbery" in m or "attack" in m:
        return "Do not resist. Call SAPS 10111 after safe."
    return "PELO Guardian. Emergency: SAPS 10111 | GBV: 0800 428 428 | Childline: 0800 055 555"

# WHATSAPP SERVICE (Mock if no Twilio)
# ============================================================
async def send_whatsapp(to: str, message: str) -> bool:
    if not TWILIO_ACCOUNT_SID:
        print(f"WhatsApp (mock): to={to}, msg={message[:50]}...")
        return True
    try:
        from twilio.rest import Client
        client = Client(TWILIO_ACCOUNT_SID, TWILIO_AUTH_TOKEN)
        client.messages.create(body=message, from_=f'whatsapp:{TWILIO_WHATSAPP_NUMBER}', to=f'whatsapp:{to}')
        return True
    except:
        return False

@app.post("/api/auth/register")
async def register(req: RegisterRequest):
    if req.email in users_db:
        raise HTTPException(status_code=400, detail="Email already registered")
    user_id = f"user_{len(users_db)+1}"
    token = jwt.encode({"sub": user_id, "name": req.name, "email": req.email, "exp": datetime.utcnow() + timedelta(days=30)}, JWT_SECRET)
    users_db[req.email] = {"userId": user_id, "name": req.name, "email": req.email, "phone": req.phone, "nextOfKinWhatsApp": req.nextOfKinWhatsApp, "securityPhone": req.securityPhone}
    return {"token": token, "user": {"userId": user_id, "name": req.name, "email": req.email}}

@app.post("/api/auth/login")
async def login(req: LoginRequest):
    if req.email not in users_db:
        raise HTTPException(status_code=401, detail="Invalid credentials")
    user = users_db[req.email]
    token = jwt.encode({"sub": user["userId"], "name": user["name"], "email": user["email"], "exp": datetime.utcnow() + timedelta(days=30)}, JWT_SECRET)
    return {"token": token, "user": user}

@app.post("/api/incidents")
async def create_incident(incident: IncidentCreate, user=Depends(get_current_user)):
    classification = await classify_incident(incident.description)
    incident_data = {
        "incidentId": f"inc_{len(incidents_db)+1}",
        "userId": user["userId"],
        "category": incident.category or classification.get("category", "other"),
        "severity": classification.get("severity", "medium"),
        "location": incident.location,
        "description": incident.description,
        "videoUrl": incident.videoUrl,
        "aiSummary": classification.get("summary", incident.description[:100]),
        "createdAt": datetime.now().isoformat()
    }
    incidents_db.insert(0, incident_data)
    return incident_data
